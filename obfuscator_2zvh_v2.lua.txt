local function rndName()
    local c="ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz"
    local s=""
    for i=1,math.random(1,3) do
        local j=math.random(1,#c)
        s=s..c:sub(j,j)
    end
    s=s.."_"
    for i=1,math.random(2,4) do s=s..string.format("%X",math.random(0,15)) end
    return s
end
local used={}
local function uniq() local n repeat n=rndName() until not used[n] used[n]=true return n end
local function shuffle(s)
    local t={} for i=1,#s do t[i]=s:sub(i,i) end
    for i=#t,2,-1 do local j=math.random(i) t[i],t[j]=t[j],t[i] end
    return table.concat(t)
end

local function bxor(a,b)
    if bit32 and bit32.bxor then return bit32.bxor(a,b) end
    a,b=math.floor(a)%256,math.floor(b)%256
    local r=0
    for i=0,7 do if a%2~=b%2 then r=r+2^i end a,b=math.floor(a/2),math.floor(b/2) end
    return r
end
local function b64enc(data,cs)
    local o,i={},1
    while i<=#data do
        local a,b,c=string.byte(data,i) or 0,string.byte(data,i+1) or 0,string.byte(data,i+2) or 0
        local n=a*65536+b*256+c
        o[#o+1]=cs:sub(math.floor(n/262144)%64+1,math.floor(n/262144)%64+1)
        o[#o+1]=cs:sub(math.floor(n/4096)%64+1,math.floor(n/4096)%64+1)
        o[#o+1]=i+1<=#data and cs:sub(math.floor(n/64)%64+1,math.floor(n/64)%64+1) or "="
        o[#o+1]=i+2<=#data and cs:sub(n%64+1,n%64+1) or "="
        i=i+3
    end
    return table.concat(o)
end
local function b85enc(data,cs)
    local base,o,i=#cs,{},1
    while i<=#data do
        local cl=math.min(4,#data-i+1)
        local n=0 for j=0,cl-1 do n=n*256+string.byte(data,i+j) end
        local ol=({[1]=2,[2]=3,[3]=4})[cl] or 5
        local e={} for k=ol,1,-1 do e[k]=cs:sub(n%base+1,n%base+1) n=math.floor(n/base) end
        for k=1,ol do o[#o+1]=e[k] end
        i=i+cl
    end
    return table.concat(o)
end
local function b91enc(data,cs)
    local s=b85enc(data,cs)
    local noise,noisy=" '\"",{}
    for i=1,#s do
        noisy[#noisy+1]=s:sub(i,i)
        if i%7==0 then local ni=math.random(1,#noise) noisy[#noisy+1]=noise:sub(ni,ni) end
    end
    return table.concat(noisy)
end
local function xorCrypt(d,k)

    return table.concat(o)
end
local function zvhEnc(data,key)
    local o,kl={},#key
    for i=1,#data do
        local kb=string.byte(key,(i-1)%kl+1)
        local b=(string.byte(data,i)+kb)%256
        b=bxor(b,(kb*7)%256)
        b=((b*3+kb)%256)
        b=((b*8)%256)+math.floor(b/32)
        b=b%256
        b=((b%16)*16+math.floor(b/16))%256
        b=bxor(b,(kb*11)%256)
        b=(b+((kb*13)%256))%256
        b=bxor(b,(kb*17)%256)
        o[i]=string.char(b)
    end
    return table.concat(o)
end

local function safeQuote(s)
    for eq=5,30 do
        local open="["..string.rep("=",eq).."["
        local close="]"..string.rep("=",eq).."]"
        if not s:find(close,1,true) then
            local q=open..s..close
            if pcall(function() return load("return "..q)() end) then return q end
        end
    end
    return string.format("%q",s)
end

local function genOpaqueTrue()
    local a,b=math.random(1,999),math.random(1,999)
    return string.format("((%d~=%d)or(%d==%d)or true)",a,b,a,a)
end
local function genOpaqueFalse()
    local a=math.random(1,999)
    return string.format("((%d==%d)and false)",a,a+1)
end
local function genOpaqueState(realNext, junkStates)
    local a = math.random(100,999)
    local b = math.random(100,999)
    local c = math.random(2,9)
    local style = math.random(1,4)
    if style == 1 then
        return string.format("((%d*%d-%d*%d)==0 and %d or %d)", a, b, a, b, realNext, junkStates[math.random(1,#junkStates)] or realNext+1)
    elseif style == 2 then
        return string.format("((%d%%2==%d%%2) and %d or %d)", a, a, realNext, realNext+math.random(10,50))
    elseif style == 3 then
        return string.format("(function() local _s=%d for _=1,%d do _s=_s end return _s end)()", realNext, c)
    else
        return string.format("((%d~=%d or true) and %d)", a, a+1, realNext)
    end
end
local function flattenControlFlow(src)
    local chunks = {}
    local buf = {}
    local depth = 0
    local function flush()
        local s = table.concat(buf, "\n")
        s = s:gsub("^%s*\n",""):gsub("\n%s*$","")
        if #s > 0 then chunks[#chunks+1] = s end
        buf = {}
    end
    local function countOpen(line)
        local d = 0
        for kw in line:gmatch("%f[%w](function)%f[%W]") do d = d + 1 end
        for kw in line:gmatch("%f[%w](do)%f[%W]") do d = d + 1 end
        for kw in line:gmatch("%f[%w](then)%f[%W]") do d = d + 1 end
        for kw in line:gmatch("%f[%w](repeat)%f[%W]") do d = d + 1 end
        for kw in line:gmatch("%f[%w](end)%f[%W]") do d = d - 1 end
        for kw in line:gmatch("%f[%w](until)%f[%W]") do d = d - 1 end
        return d
    end
    for line in (src.."\n"):gmatch("(.-)\n") do
        buf[#buf+1] = line
        local delta = countOpen(line)
        depth = depth + delta
        if depth < 0 then depth = 0 end
        if depth == 0 then
            local trimmed = line:gsub("^%s+",""):gsub("%s+$","")
            if trimmed:match("^end%s*$") or trimmed:match("^until%s+") or
               trimmed:sub(-1) == ";" or
               trimmed:match("^return%f[%W]") or trimmed:match("^break%f[%W]") or
               (trimmed:match("^local%s+") and not trimmed:match("function")) or
               (trimmed:match("^=%s*") and not trimmed:match("function")) then
                if #buf >= 1 then
                    flush()
                end
            end
        end
    end
    flush()
    if #chunks < 2 then
        chunks = { src }
    end
    local MAX_BLOCKS = 24
    if #chunks > MAX_BLOCKS then
        local merged = {}
        local per = math.ceil(#chunks / MAX_BLOCKS)
        for i=1,#chunks,per do
            local part = {}
            for j=i,math.min(i+per-1,#chunks) do part[#part+1] = chunks[j] end
            merged[#merged+1] = table.concat(part, "\n")
        end
        chunks = merged
    end
    local n = #chunks
    local states, usedStates = {}, {}
    local function rndState()
        local s
        repeat s = math.random(1000, 99999) until not usedStates[s]
        usedStates[s] = true
        return s
    end
    for i=1,n do states[i] = rndState() end
    local entryState = states[1]
    local exitState  = rndState()
    local junkCount = math.min(6, math.max(3, math.floor(n * 0.4)))
    local junkStates = {}
    for i=1,junkCount do junkStates[i] = rndState() end
    local stateVar = uniq()
    local nextVar  = uniq()
    local contVar  = uniq()
    local parts = {
        string.format("local %s=%d local %s=true", stateVar, entryState, contVar),
        string.format("while %s do", contVar),
        string.format("  local %s=%s", nextVar, stateVar),
    }
    for i=1,n do
        local nxt
        if i < n then
            nxt = genOpaqueState(states[i+1], junkStates)
        else
            nxt = string.format("(function() %s=false return %d end)()", contVar, exitState)
        end
        if i == 1 then
            parts[#parts+1] = string.format("  if %s==%d then", stateVar, states[i])
        else
            parts[#parts+1] = string.format("  elseif %s==%d then", stateVar, states[i])
        end
        for line in (chunks[i].."\n"):gmatch("(.-)\n") do
            parts[#parts+1] = "    "..line
        end
        parts[#parts+1] = string.format("    %s=%s", nextVar, nxt)
    end
    for i=1,junkCount do
        local js = junkStates[i]
        local fakeNext = junkStates[((i % junkCount) + 1)]
        parts[#parts+1] = string.format("  elseif %s==%d then", stateVar, js)
        local jv = uniq()
        parts[#parts+1] = string.format("    local %s=0", jv)
        parts[#parts+1] = string.format("    for _=1,%d do %s=%s+math.random(1,5) end", math.random(3,8), jv, jv)
        if math.random() > 0.5 then
            parts[#parts+1] = string.format("    if %s then %s=%s end", genOpaqueFalse(), jv, jv)
        end
        parts[#parts+1] = string.format("    %s=%s", nextVar, genOpaqueState(fakeNext, junkStates))
    end
    parts[#parts+1] = "  else"
    parts[#parts+1] = string.format("    %s=false", contVar)
    parts[#parts+1] = "  end"
    parts[#parts+1] = string.format("  %s=%s", stateVar, nextVar)
    parts[#parts+1] = "end"
    return table.concat(parts, "\n")
end
local function packNames(list, key)
    key = key or 91
    local joined = table.concat(list, "|")
    local t = {}
    for i = 1, #joined do
        local a, b = string.byte(joined, i), key
        local r = 0
        for j = 0, 7 do
            if a % 2 ~= b % 2 then r = r + 2 ^ j end
            a = math.floor(a / 2)
            b = math.floor(b / 2)
        end
        t[i] = r
    end
    return table.concat(t, ","), key
end

local function genJunk()

        local i = math.random(1, #letters)

    end

        return rndLetter() .. rndLetter()
    end
    local function rndNum(a, b) return math.random(a or 0, b or 9999) end
    local v1, v2, v3 = uniq(), uniq(), uniq()
    if style == 1 then

            else parts[#parts+1]=string.format("[%q]=%d", rndKey(), rndNum()) end
        end
        return string.format("%s={%s}", name, table.concat(parts, ","))
    else
        return string.format("%s=function()local %s,%s=pcall(function() return %d end) return %s and %s or %d end", name, v1, v2, rndNum(), v1, v2, rndNum())
    end
end
local function genPadBlob(targetBytes)
    local parts = {}
    local size = 0
    local syms = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789"
    local letters = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    local function rl() local i=math.random(1,#letters) return letters:sub(i,i) end
    while size < targetBytes do
        local name = uniq()
        local style = math.random(1, 8)
        local a, b, c = uniq(), uniq(), uniq()
        local chunk
        if style == 1 then
            local n = math.random(40, 100)
            local nums = {}
            for i=1,n do nums[i] = tostring(math.random(0,999999)) end
            chunk = string.format("%s={%s}", name, table.concat(nums, ","))
        elseif style == 2 then
            chunk = string.format("%s=function()local %s=%d while %s<%d do %s=%s+%d if %s>%d then break end end return %s end",
                name, a, math.random(0,3), a, math.random(10,30), a, a, math.random(1,3), a, math.random(15,40), a)
        elseif style == 3 then
            chunk = string.format("%s=function(%s)local %s=0 for %s=1,%d do if %s%%%d==0 then %s=%s+%s else %s=%s+1 end end return %s end",
                name, a, b, c, math.random(5,20), c, math.random(2,5), b, b, c, b, b, b)
        elseif style == 4 then
            chunk = string.format("%s=(function() do local %s=%d local %s={%d,%d,%d} for i=1,#%s do %s=%s+%s[i] end return %s end end)()",
                name, a, math.random(1,100), b, math.random(1,50), math.random(1,50), math.random(1,50), b, a, a, b, a)
        elseif style == 5 then
            local n = math.random(20, 60)
            local keys = {}
            for i=1,n do keys[i] = string.format("[%q]=%d", rl()..rl(), math.random(0,999999)) end
            chunk = string.format("%s={%s}", name, table.concat(keys, ","))
        elseif style == 6 then
            chunk = string.format("%s=function()local %s=0 repeat %s=%s+1 local %s=%s*%d until %s>=%d return %s end",
                name, a, a, a, b, a, math.random(2,5), a, math.random(8,20), b)
        elseif style == 7 then
            local body = {}
            for i=1,math.random(20,50) do local j=math.random(1,#syms) body[i]=syms:sub(j,j) end
            chunk = string.format("%s=function()local %s=%s local %s=#%s return %s>%d and %s or %d end",
                name, a, safeQuote(table.concat(body)), b, a, b, math.random(5,30), b, math.random(0,99))
        else
            chunk = string.format("%s=function(%s,%s) if %s>%s then return %s*%d elseif %s==%s then return %d else local %s=%s+%s return %s end end",
                name, a, b, a, b, a, math.random(2,9), a, b, math.random(0,999), c, a, b, c)
        end
        parts[#parts+1] = chunk
        size = size + #chunk + 1
    end
    return parts
end
